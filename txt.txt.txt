CREATE TABLE fakecustomerinfo.default.fake_customer_data_brasil AS
SELECT 
    placeholder,
    CASE 
        WHEN name = 'David' THEN 'Davi'
        WHEN name = 'John' THEN 'João'
        WHEN name = 'Tiffany' THEN 'Tifany'
        WHEN name = 'Greg' THEN 'Gregório'
        ELSE name 
    END AS nome_brasileiro,
    status,
    flight,
    (cost_usd * 5) AS custo_brl,
    distance_km,
    CASE 
        WHEN region = 'NA' THEN 'BR'
        ELSE region 
    END AS regiao_atualizada
FROM 
    fakecustomerinfo.default.fake_customer_data;
