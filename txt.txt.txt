-- Passo 1: Criar a tabela com a estrutura brasileira vazia
CREATE TABLE fakecustomerinfo.default.fake_customer_data_brasil (
    placeholder VARCHAR,
    nome_brasileiro VARCHAR,
    status VARCHAR,
    flight VARCHAR,
    custo_brl BIGINT,
    distance_km BIGINT,
    regiao_atualizada VARCHAR
);

-- Passo 2: Inserir os dados transformados para o contexto brasileiro
INSERT INTO fakecustomerinfo.default.fake_customer_data_brasil
SELECT 
    placeholder,
    CASE 
        WHEN name = 'David' THEN 'Davi'
        WHEN name = 'John' THEN 'João'
        WHEN name = 'Tiffany' THEN 'Tifany'
        WHEN name = 'Greg' THEN 'Gregório'
        ELSE name 
    END,
    status,
    flight,
    (cost_usd * 5),
    distance_km,
    CASE 
        WHEN region = 'NA' THEN 'BR'
        ELSE region 
    END

    fakecustomerinfo.default.fake_customer_data;
